---대화 시스템 구문 정리---

1. 전체 구문 템플릿
{n{as cn}}{ at x{:y}}{ [1:{ + 1:e}]} "d"} {c(a){, c(a)}}

- n: 대화 번호/순서
- cn: 캐릭터 이름 지정
- x:y: 위치 좌표
- d: 실제 대사 내용
- c(a): 명령어와 파라미터

2. 실제 예시 분석
Elen "Let's listen to some music!" PlaySong("Ethereal Presence" -v 1 -p 0.3)
5                                  33                          34

Navigator "Captain, we're approaching the planet."
5         33

Captain "What kind of readings?" faceLeft(Captain), move(Captain 0.7)

5. 특수 문자 처리
- 대사 안의 따옴표: \" 사용
 예시: 캐릭터 "그가 \"안녕\"이라고 말했어요"
- 여러 명령어는 쉼표(,)로 구분
 예시: fadeIn(1), wait(0.5), PlaySong("곡이름")
- 파라미터는 공백으로 구분
 예시: move(캐릭터 0.7)

using UnityEngine;
using System.Collections.Generic;

// 1. 플래그 시스템 - 게임의 상태와 변수들을 관리
public class FlagSystem
{
   // 불리언 플래그를 저장하는 딕셔너리 (예: 아이템 획득 여부, 이벤트 발생 여부)
   private Dictionary<string, bool> flags = new Dictionary<string, bool>();
   // 수치형 변수를 저장하는 딕셔너리 (예: 호감도, 스탯)
   private Dictionary<string, float> variables = new Dictionary<string, float>();
   
   // 플래그 설정/획득 메서드
   public void SetFlag(string key, bool value) => flags[key] = value;
   public bool GetFlag(string key) => flags.GetValueOrDefault(key, false);
   
   // 변수 설정/획득 메서드
   public void SetVariable(string key, float value) => variables[key] = value;
   public float GetVariable(string key) => variables.GetValueOrDefault(key, 0f);
}

// 2. 노드 시스템 - 복잡한 대화 분기와 선택지를 관리
// 대화 노드 클래스 - 하나의 대화 상황을 표현
public class DialogueNode
{
   public string nodeID;        // 노드 식별자
   public string text;          // 대화 내용
   public List<Choice> choices; // 선택지 목록
   public Dictionary<string, object> conditions;  // 노드 실행 조건
   public Dictionary<string, object> effects;     // 노드 실행 효과
}

// 선택지 클래스 - 플레이어가 선택할 수 있는 옵션
public class Choice
{
   public string text;          // 선택지 텍스트
   public string nextNode;      // 선택 후 이동할 노드
   public Dictionary<string, object> conditions;  // 선택지 표시 조건
   public Dictionary<string, object> effects;     // 선택 시 발생할 효과
}

// 노드 관리 시스템
public class NodeSystem
{
   private Dictionary<string, DialogueNode> nodes;  // 모든 대화 노드 저장
   private FlagSystem flagSystem;                   // 플래그 시스템 참조

   // 특정 노드부터 대화 시작
   public void StartNode(string nodeName)
   {
       var node = nodes[nodeName];
       // 노드 실행 조건 체크
       if(node.conditions == null || CheckConditions(node.conditions))
       {
           ProcessNode(node);
       }
   }

   // 노드 처리 로직
   private void ProcessNode(DialogueNode node)
   {
       ShowDialogue(node.text);  // 대화 표시
       // 선택지가 있는 경우 표시
       if(node.choices != null && node.choices.Count > 0)
       {
           ShowChoices(node.choices);
       }
   }
}

// 3. 스크립트 시스템 - 일반적인 대화 진행을 관리
public class ScriptSystem 
{
   private FlagSystem flagSystem;    // 플래그 시스템 참조
   private NodeSystem nodeSystem;    // 노드 시스템 참조

   // 한 줄의 스크립트 처리
   public void ProcessLine(string line)
   {
       // 노드 분기 명령어 처리
       if(line.StartsWith("[branch"))
       {
           string nodeName = ExtractNodeName(line);
           nodeSystem.StartNode(nodeName);
       }
       // 조건문 처리
       else if(line.StartsWith("[if"))
       {
           ProcessCondition(line);
       }
       // 일반 대화 처리
       else
       {
           ProcessDialogue(line);
       }
   }
}

// 4. 메인 관리자 - 모든 시스템을 통합 관리
public class DialogueManager : MonoBehaviour
{
   private NodeSystem nodeSystem;
   private FlagSystem flagSystem;
   private ScriptSystem scriptSystem;
   
   // 현재 진행 모드 (스크립트 또는 노드)
   private enum Mode { Script, Node }
   private Mode currentMode = Mode.Script;

   // 시스템 초기화
   void Start()
   {
       flagSystem = new FlagSystem();
       nodeSystem = new NodeSystem();
       scriptSystem = new ScriptSystem();
   }

   // 스토리 진행 시작
   public void ProcessStory(string storyFile)
   {
       scriptSystem.StartProcessing(storyFile);
   }
}
